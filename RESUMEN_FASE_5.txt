OPCIÓN 2: INTEGRAR EN TU CÓDIGO
──────────────────────────────
from agents.followup_agent import FollowUpAgent

agent = FollowUpAgent(orchestrator)

# Iniciar seguimiento cuando envías propuesta
agent.start_followup_sequence(client_id=1, proposal_id=101)

# Cada hora (o con scheduler):
pending = agent.get_pending_followups()
for item in pending:
    agent.send_followup(item['sequence_id'], item['followup_number'])

# Cuando cliente abre email (webhook):
agent.mark_followup_opened(sequence_id=1, followup_number=1)

# Cuando cliente responde:
agent.complete_sequence(sequence_id=1, reason="cliente_respondió")

# Ver estado:
status = agent.get_followup_status(client_id=1)
